  \begin{pblock}{1. Motivación}
    \begin{itemize}
      \item Comunidad \textbf{qom} (toba); $\sim$80.000 hablantes.
      \item Región del \textbf{Gran Chaco} (norte argentino), migración a
            grandes ciudades.
      \item Lengua con uso en \textbf{retracción} y \textbf{escasos recursos digitales}.
      \item Traducción automática qom\,$\leftrightarrow$\,español como recurso valioso para \textbf{documentar y revitalizar} la lengua.
      \item Confiar en un traductor requiere \textbf{evaluarlo}: ¿cómo sabemos si sus traducciones son \textbf{adecuadas}?
    \end{itemize}
  \end{pblock}

  \begin{pblock}{3. El problema}
    \begin{itemize}
      \item ¿\textbf{Cómo dar cuenta de la calidad de las traducciones} automáticas? \textbf{Métricas de evaluación}.
      \item \textbf{¿Cuál es el desafío?} \textbf{Métricas existentes poco robustas} o con limitaciones en lenguas de bajos recursos.
    \end{itemize}
    \vspace{0.4cm}
    \begin{center}
    \small
    \renewcommand{\arraystretch}{1.25}
    \begin{tabularx}{\linewidth}{l >{\raggedright\arraybackslash}X >{\raggedright\arraybackslash}X}
      \toprule
      \textbf{Métrica} & \textbf{Descripción} & \textbf{Limitación} \\
      \midrule
      \multicolumn{3}{c}
          {\textbf{\textit{Basadas en solapamiento de n-gramas entre texto esperado y predicho}}} \\
          BLEU\cita{3}
              & Uso de n-gramas de palabras.
              & Penalización de variación léxica y morfológica. \\
          ChrF++\textsuperscript{a}\cita{4}
              & Uso de n-gramas de caracteres (y palabras).
              & Mayor robustez para morfología rica; penalización de la variación ortográfica. \\
      \midrule
      \multicolumn{3}{c}
          {\textbf{\textit{Basadas en embeddings para representar texto esperado y predicho}}} \\
          \begin{tabular}[t]{@{}l@{}}BERTScore\cita{5}\\MoverScore\cita{6}\\COMET\cita{7}\end{tabular}
              & Representación vectorial y cálculo de similitud entre ambos textos. Cada métrica presenta variaciones propias.
              & Necesidad de un modelo pre-entrenado con gran cantidad de datos, difícil de obtener en contexto de bajos recursos. \\
      \bottomrule
    \end{tabularx}
    \end{center}
    \vspace{0.2cm}
      {\small \textsuperscript{a} Estándar reportado actualmente en competencias de
    NLP de lenguas de bajos recursos.\par}
    \begin{phighlight}
      AmericasNLP 2025\cita[white]{8}: correlación \textit{baseline} con juicios humanos
      \boldmath ChrF++ $r$=$\sim$0.6, $\rho$=$\sim$0.52
    \end{phighlight}
  \end{pblock}

  \begin{pblock}{4. Propuesta}
  Proponemos:
    \begin{itemize}
      \item Desarrollar \textbf{métricas de evaluación}:
      {\setbeamerfont{itemize/enumerate subbody}{size=\normalsize}
      \setbeamertemplate{itemize subitem}
      {\color{tudBlue}$\blacktriangleright$}
      \begin{itemize}
        \item \textbf{anotaciones humanas}
        \item \textbf{técnicas automáticas}
      \end{itemize}
      }
      \item \textbf{Correlacionar} ambos tipos de métricas y \textbf{analizar} esa correlación.
      \end{itemize}
      \textbf{Aspecto relevante:} particularidades lingüísticas $\rightarrow$ impacto en tipos de error
      {\setbeamertemplate{itemize item}{\color{black}$\bullet$}
      \begin{itemize}
        \item tipología de la lengua
        \item riqueza morfológica
        \item nivel de estandarización ortográfica
      \end{itemize}
      }
  \end{pblock}

  \begin{pblock}{5. Métricas a partir de evaluaciones humanas}
    \begin{itemize}
      \item \textbf{Estatus:} elaboración del criterio de anotación en curso. | \textbf{\textit{Framework}} candidato: \textbf{MQM}\cita{9}.
      \item \textbf{Grupo de anotadores:}
      {\setbeamerfont{itemize/enumerate subbody}{size=\normalsize}
       \setbeamertemplate{itemize subitem}{\color{tudBlue}$\blacktriangleright$}
       \setlength{\multicolsep}{0pt}
       \begin{multicols}{2}
         \begin{itemize}
           \item homogéneo: reduce el riesgo de acuerdo bajo entre anotadores
           \item con conocimiento técnico de la lengua
           \item con dominio robusto de ambas lenguas
           \item perteneciente a la comunidad qom
         \end{itemize}
       \end{multicols}
      }
      \item \textbf{¿Mismos criterios en ambos sentidos?} Flexibilizar la ortografía en qom (sin estandarización plena); contemplar terminología cultural sin traducción unívoca.
      \item \textbf{Aspecto relevante:} factores socioculturales \,$\rightarrow$\, impacto en soporte de anotación, formato.
    \end{itemize}
    \vspace{0.2cm}
    {\bfseries Ejemplos (anotación inicial, 28 segmentos por dirección, tipo de error MQM más cercano entre paréntesis asignado \underline{\textit{a posteriori}} de la anotación):}
    \begin{pinner}
      \small
      \textbf{qom\,$\rightarrow$\,español}
      \begin{itemize}
        \item Sustitución léxica + alucinación (MQMN\_111100): ``nido de guazuncho y cuero'' en lugar de ``gallineros y corrales''.
        \item Desajuste sintáctico / estructura de la información (MQMN\_201030): ``para los qom'' traducido como ``esa gente toba''.
      \end{itemize}
      \textbf{español\,$\rightarrow$\,qom}
      \begin{itemize}
        \item Limitación morfosintáctica (MQMN\_201020): orden de palabras inadecuado, p.\,ej.\ la posición de ``ndoteec'' (``solamente'') en ``Ndoteec da a\~{y}em 'auchoxoden'' es semánticamente comprensible pero gramaticalmente incorrecta.
      \end{itemize}
    \end{pinner}
  \end{pblock}

  \begin{pblock}{6. Métricas automáticas y primeros avances}
    \textbf{En curso:}
    \begin{itemize}
      \item \textbf{Basadas en n-gramas:}
      {\setbeamerfont{itemize/enumerate subbody}{size=\normalsize}
      \setbeamertemplate{itemize subitem}
      {\color{tudBlue}$\blacktriangleright$}
      \begin{itemize}
           \item evaluación de modelos con valores estándares y alternativos de parámetros
           \item correlacionar con evaluaciones realizadas por humanos
      \end{itemize}
      }
      \item \textbf{Basadas en \textit{embeddings}:}
      {\setbeamerfont{itemize/enumerate subbody}{size=\normalsize}
      \setbeamertemplate{itemize subitem}
      {\color{tudBlue}$\blacktriangleright$}
      \begin{itemize}
           \item qué datos (digitalización de nuevas fuentes)
           \item qué particiones (estratificación, evitar el uso de Biblia en \textit{test} y \textit{data leakage})
           \item cuál es el rendimiento sin \textit{fine-tuning}
           \item qué modelo base; referencias \,$\rightarrow$\, AmericasNLP 2025: \textbf{LaBSE}\cita{10} + [ \textbf{YiSi-1}\cita{11} o \textbf{COMET} o esquemas de \textbf{similitud/regresión} ] (Krasner et al.\cita{12}; Raja \& Vats\cita{13}).
      \end{itemize}
      }
